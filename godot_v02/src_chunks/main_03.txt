func glow_box(color:Color, radius:float) -> StyleBoxFlat:
	var s := StyleBoxFlat.new()
	s.bg_color = color
	s.corner_radius_top_left = int(radius); s.corner_radius_top_right=int(radius); s.corner_radius_bottom_left=int(radius); s.corner_radius_bottom_right=int(radius)
	s.border_width_left=2; s.border_width_right=2; s.border_width_top=2; s.border_width_bottom=2
	s.border_color = Color(color.r+0.15,color.g+0.15,color.b+0.15,min(1.0,color.a+0.12))
	return s
